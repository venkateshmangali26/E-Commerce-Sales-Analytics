   192→export function weightedPick<T>(items: readonly T[], weights: number[]): T {
   193→  const total = weights.reduce((s, w) => s + w, 0);
   194→  let r = Math.random() * total;
   195→  for (let i = 0; i < items.length; i++) {
   196→    r -= weights[i];
   197→    if (r <= 0) return items[i];
   198→  }
   199→  return items[items.length - 1];
   200→}
   201→
   202→export function makeRng(seed: number) {
   203→  let t = seed >>> 0;
   204→  return function () {
   205→    t = (t + 0x6d2b79f5) | 0;
   206→    let x = Math.imul(t ^ (t >>> 15), 1 | t);
   207→    x = (x + Math.imul(x ^ (x >>> 7), 61 | x)) ^ x;
   208→    return ((x ^ (x >>> 14)) >>> 0) / 4294967296;
   209→  };
   210→}

   253→function pearson(xs: number[], ys: number[]) {
   254→  const n = xs.length;
   255→  if (n === 0) return 0;
   256→  const mx = xs.reduce((s, x) => s + x, 0) / n;
   257→  const my = ys.reduce((s, x) => s + x, 0) / n;
   258→  let num = 0, dx = 0, dy = 0;
   259→  for (let i = 0; i < n; i++) {
   260→    const a = xs[i] - mx, b = ys[i] - my;
   261→    num += a * b; dx += a * a; dy += b * b;
   262→  }
   263→  const denom = Math.sqrt(dx) * Math.sqrt(dy);
   264→  return denom === 0 ? 0 : num / denom;
   265→}

   279→export function computeAnalytics(orders: Order[]): AnalyticsPayload {
   280→  let totalSales = 0, totalProfit = 0, totalShipping = 0, totalDiscount = 0, negProfit = 0;
   281→  const customers = new Set<string>();
   282→  const yearly: Record<number, { revenue: number; profit: number; orders: number }> = {};
   283→
   284→  for (const o of orders) {
   285→    totalSales += o.totalSales;
   286→    totalProfit += o.profit;
   287→    totalShipping += o.shippingCost;
   288→    totalDiscount += o.discountPercent;
   289→    if (o.profit < 0) negProfit++;
   290→    customers.add(o.customerName);
   291→    const y = o.orderDate.getUTCFullYear();
   292→    if (!yearly[y]) yearly[y] = { revenue: 0, profit: 0, orders: 0 };
   293→    yearly[y].revenue += o.totalSales;
   294→    yearly[y].profit += o.profit;
   295→    yearly[y].orders += 1;
   296→  }
   297→  const totalOrders = orders.length;
   298→
   299→  const kpis = {
   300→    total_sales: fcur(totalSales),
   301→    total_profit: fcur(totalProfit),
   302→    total_orders: totalOrders,
   303→    avg_order_value: fcur(totalSales / totalOrders),
   304→    avg_discount: round(totalDiscount / totalOrders, 2),
   305→    profit_margin: round((totalProfit / totalSales) * 100, 2),
   306→    total_shipping: fcur(totalShipping),
   307→    unique_customers: customers.size,
   308→    negative_profit_orders: negProfit,
   309→    yearly: Object.fromEntries(
   310→      Object.entries(yearly).map(([k, v]) => [Number(k), {
   311→        revenue: fcur(v.revenue), profit: fcur(v.profit), orders: v.orders,
   312→      }])
   313→    ),
   314→  };
   315→
   316→  // Monthly
   317→  const monthlyMap = new Map<string, { revenue: number; profit: number; orders: number }>();
   318→  for (const o of orders) {
   319→    const ym = `${o.orderDate.getUTCFullYear()}-${String(o.orderDate.getUTCMonth() + 1).padStart(2, "0")}`;

   329→    });
   330→
   331→  const yoy = Object.entries(yearly).map(([k, v]) => ({
   332→    year: Number(k), revenue: fcur(v.revenue), profit: fcur(v.profit), orders: v.orders,
   333→  })).sort((a, b) => a.year - b.year);
   334→

   431→  const discount_scatter = sample.map((o) => ({
   432→    discount: o.discountPercent,
   433→    profit_margin: o.totalSales > 0 ? round((o.profit / o.totalSales) * 100, 2) : 0,
   434→    total_sales: fcur(o.totalSales), category: o.productCategory, product: o.productName,
   435→  }));
   436→
   437→  // Discount ranges
   438→  const drMap = new Map<string, { orders: number; revenue: number; profit: number; marginSum: number }>();
   439→  for (const o of orders) {
   440→    const r = discountRange(o.discountPercent);
   441→    if (!drMap.has(r)) drMap.set(r, { orders: 0, revenue: 0, profit: 0, marginSum: 0 });
   442→    const e = drMap.get(r)!;
   443→    e.orders += 1; e.revenue += o.totalSales; e.profit += o.profit;
   444→    if (o.totalSales > 0) e.marginSum += (o.profit / o.totalSales) * 100;
   445→  }
   446→  const drOrder = ["No Discount", "1-10%", "11-20%", "21-35%"];
   447→  const discount_ranges = drOrder.map((r) => {
   448→    const d = drMap.get(r)!;
   449→    return { range: r, orders: d.orders, avg_revenue: fcur(d.revenue / d.orders),
   450→      avg_profit: fcur(d.profit / d.orders), avg_margin: round(d.marginSum / d.orders, 2) };
   451→  }).filter((d) => d.orders > 0);
   452→
   453→  // Payment
   454→  const payMap = new Map<string, { orders: number; revenue: number }>();
   455→  for (const o of orders) {
   456→    if (!payMap.has(o.paymentMethod)) payMap.set(o.paymentMethod, { orders: 0, revenue: 0 });
   457→    const e = payMap.get(o.paymentMethod)!;
   458→    e.orders += 1; e.revenue += o.totalSales;
   459→  }
   460→  const totalPayOrders = Array.from(payMap.values()).reduce((s, v) => s + v.orders, 0);
   461→  const payment = Array.from(payMap.entries())
   462→    .map(([k, v]) => ({
   463→      method: k, orders: v.orders, revenue: fcur(v.revenue),
   464→      share: round((v.orders / totalPayOrders) * 100, 1), avg_order: fcur(v.revenue / v.orders),
   465→    })).sort((a, b) => b.orders - a.orders);
   466→
   467→  const shipping = region.map((r) => ({
   468→    region: r.region, avg_shipping: r.avg_shipping, profit_margin: r.profit_margin,
   469→  }));
   470→
   471→  // Correlation
   472→  const CORR_FEATURES = ["Quantity", "Unit_Price", "Discount_Percent", "Total_Sales", "Shipping_Cost", "Profit", "Profit_Margin"];
   473→  const corrCols: Record<string, number[]> = {};
   474→  for (const f of CORR_FEATURES) corrCols[f] = [];
   475→  for (const o of orders) {
   476→    corrCols.Quantity.push(o.quantity);
   477→    corrCols.Unit_Price.push(o.unitPrice);
   478→    corrCols.Discount_Percent.push(o.discountPercent);
   479→    corrCols.Total_Sales.push(o.totalSales);
   480→    corrCols.Shipping_Cost.push(o.shippingCost);
   481→    corrCols.Profit.push(o.profit);
   482→    corrCols.Profit_Margin.push(o.totalSales > 0 ? (o.profit / o.totalSales) * 100 : 0);
   483→  }
   484→  const correlation = CORR_FEATURES.map((a) => {
   485→    const row: Record<string, string | number> = { feature: a };
   486→    for (const b of CORR_FEATURES) row[b] = round(pearson(corrCols[a], corrCols[b]), 2);
   487→    return row;
   488→  });
   489→
   490→  // Day of week
   491→  const dowMap = new Map<string, { orders: number; revenue: number }>();
   492→  for (const d of DOW_ORDER) dowMap.set(d, { orders: 0, revenue: 0 });
   493→  for (const o of orders) {
   494→    const d = DOW_ORDER[o.orderDate.getUTCDay()];
   495→    const e = dowMap.get(d)!;
   496→    e.orders += 1; e.revenue += o.totalSales;
   497→  }
   498→  const dow = DOW_ORDER.slice(1).concat([DOW_ORDER[0]]).map((d) => {
   499→    const e = dowMap.get(d)!;
   500→    return { day: d, orders: e.orders, avg_revenue: e.orders > 0 ? fcur(e.revenue / e.orders) : 0 };
   501→  });
   502→
   503→  return {
   504→    kpis, monthly, yoy, heatmap, region, country, category, top_products,
   505→    segment, segment_category, discount_scatter, discount_ranges, payment,
   506→    shipping, correlation, correlation_features: CORR_FEATURES, dow,
   507→  };
   508→}
   509→
   510→export type RecentOrdersParams = {
   511→  search?: string; category?: string; region?: string;
   512→  payment?: string; segment?: string; limit?: number; offset?: number;
   513→};
   514→
   515→export async function getRecentOrders(params: RecentOrdersParams) {
   516→  const { search, category, region, payment, segment, limit = 25, offset = 0 } = params;
   517→  const where: Record<string, unknown> = {};

   521→  if (segment && segment !== "all") where.customerSegment = segment;

   523→    where.OR = [
   524→      { id: { contains: search } },
   525→      { customerName: { contains: search } },
   526→      { productName: { contains: search } },
   527→      { country: { contains: search } },
   528→    ];
   529→  }
   530→  const [total, rows] = await Promise.all([
   531→    db.order.count({ where }),
   532→    db.order.findMany({ where, orderBy: { orderDate: "desc" }, take: limit, skip: offset }),
   533→  ]);
   534→  return {
   535→    total,
   536→    rows: rows.map((o) => ({
   537→      id: o.id, date: o.orderDate.toISOString().slice(0, 10),
   538→      customer: o.customerName, country: o.country, region: o.region,
   539→      product: o.productName, category: o.productCategory, quantity: o.quantity,
   540→      total_sales: round(o.totalSales, 2), profit: round(o.profit, 2),
   541→      discount_percent: o.discountPercent, payment_method: o.paymentMethod,
   542→      customer_segment: o.customerSegment,
   543→    })),
   544→  };
   545→}

   550→### 6. [`scripts/seed-orders.ts`](file:///c:/Users/manga/Downloads/archive/scripts/seed-orders.ts) — Synthetic Data Seeder
   551→
   552→```ts
   553→import { PrismaClient } from "@prisma/client";
   554→import {
   555→  COUNTRIES, REGIONS, CATEGORIES, PRODUCTS, CUSTOMER_SEGMENTS, PAYMENT_METHODS,
   556→  REGION_SHIP_BASE, CATEGORY_MARGIN, SEGMENT_DISCOUNT, MONTH_WEIGHTS,
   557→  FIRST_NAMES, LAST_NAMES, weightedPick, makeRng,
   558→  type Region, type Category, type CustomerSegment,
   559→} from "../src/lib/ecommerce-domain";
   560→
   561→const db = new PrismaClient();
   562→const N_TXNS = 2000;
   563→const YEARS = [2023, 2024, 2025];
   564→
   565→const rand = (rng: () => number, min: number, max: number) => min + rng() * (max - min);
   566→const randInt = (rng: () => number, lo: number, hi: number) => Math.floor(rand(rng, lo, hi + 1));
   567→const pick = <T,>(arr: readonly T[], rng: () => number): T => arr[Math.floor(rng() * arr.length)];
   568→
   569→function generateOrder(i: number, rng: () => number) {
   570→  const month = weightedPick(Array.from({ length: 12 }, (_, k) => k + 1), MONTH_WEIGHTS);
   571→  const year = pick(YEARS, rng);
   572→  const day = randInt(rng, 1, 28);
   573→  const orderDate = new Date(year, month - 1, day);
   574→
   575→  const region = weightedPick(REGIONS as unknown as Region[], [34, 28, 22, 9, 7]);
   576→  const country = pick(COUNTRIES[region], rng);
   577→  const category = weightedPick(CATEGORIES as unknown as Category[], [40, 28, 20, 12]);
   578→  const product = pick(PRODUCTS[category], rng);
   579→  const unitPrice = product.price;
   580→  const quantity = weightedPick([1, 2, 3, 4, 5], [45, 28, 14, 8, 5]);
   581→  const segment = weightedPick(CUSTOMER_SEGMENTS as unknown as CustomerSegment[], [55, 30, 15]);
   582→
   583→  let [dLo, dHi] = SEGMENT_DISCOUNT[segment];
   584→  if (unitPrice > 200) dLo += 2;
   585→  const discountPercent = rng() > 0.15 ? Math.round(rand(rng, dLo, dHi) * 10) / 10 : 0;
   586→
   587→  const gross = unitPrice * quantity;
   588→  const discountAmount = (gross * discountPercent) / 100;
   589→  const totalSales = Math.round((gross - discountAmount) * 100) / 100;
   590→
   591→  const baseMargin = CATEGORY_MARGIN[category];
   592→  const shippingCost = Math.round((REGION_SHIP_BASE[region] + rand(rng, -3, 5)) * 100) / 100;
   593→  const profit = Math.round((baseMargin * gross - discountAmount - shippingCost) * 100) / 100;
   594→
   595→  const payment = weightedPick(PAYMENT_METHODS as unknown as string[], [55, 25, 12, 8]);
   596→  const customer = `${pick(FIRST_NAMES, rng)} ${pick(LAST_NAMES, rng)}`;
   597→  const yearIdx = YEARS.indexOf(year);
   598→  const id = `ORD-${yearIdx * 10000 + i + 1}`;
   599→
   600→  return {
   601→    id, orderDate, customerName: customer, customerSegment: segment,
   602→    country, region, productCategory: category, productName: product.name,
   603→    quantity, unitPrice, discountPercent, shippingCost, totalSales,
   604→    profit, paymentMethod: payment,
   605→  };
   606→}
   607→
   608→async function main() {
   609→  const existing = await db.order.count();
   610→  if (existing > 0) {
   611→    console.log(`Database already has ${existing} orders. Skipping seed.`);
   612→    return;
   613→  }
   614→  console.log(`Seeding ${N_TXNS} synthetic orders...`);
   615→  const rng = makeRng(42);
   616→  const CHUNK = 250;
   617→  for (let chunkStart = 0; chunkStart < N_TXNS; chunkStart += CHUNK) {
   618→    const batch = [];
   619→    for (let i = chunkStart; i < Math.min(chunkStart + CHUNK, N_TXNS); i++) {
   620→      batch.push(generateOrder(i, rng));
   621→    }
   622→    await db.order.createMany({ data: batch });
   623→    console.log(`  Inserted orders ${chunkStart + 1}–${chunkStart + batch.length}`);
   624→  }
   625→  const final = await db.order.count();
   626→  console.log(`✓ Seed complete. Database now has ${final} orders.`);
   627→}
   628→
   629→main().catch((e) => { console.error("Seed failed:", e); process.exit(1); })
   630→  .finally(async () => { await db.$disconnect(); });
   631→```
