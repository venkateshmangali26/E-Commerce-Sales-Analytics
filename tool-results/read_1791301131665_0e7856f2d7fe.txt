   217→export function weightedPick<T>(items: readonly T[], weights: number[]): T {
   218→  const total = weights.reduce((s, w) => s + w, 0);
   219→  let r = Math.random() * total;
   220→  for (let i = 0; i < items.length; i++) {
   221→    r -= weights[i];
   222→    if (r <= 0) return items[i];
   223→  }
   224→  return items[items.length - 1];
   225→}

   227→export function makeRng(seed: number) {
   228→  let t = seed >>> 0;
   229→  return function () {
   230→    t = (t + 0x6d2b79f5) | 0;
   231→    let x = Math.imul(t ^ (t >>> 15), 1 | t);
   232→    x = (x + Math.imul(x ^ (x >>> 7), 61 | x)) ^ x;
   233→    return ((x ^ (x >>> 14)) >>> 0) / 4294967296;
   234→  };
   235→}

   279→function pearson(xs: number[], ys: number[]) {
   280→  const n = xs.length;
   281→  if (n === 0) return 0;
   282→  const mx = xs.reduce((s, x) => s + x, 0) / n;
   283→  const my = ys.reduce((s, x) => s + x, 0) / n;
   284→  let num = 0, dx = 0, dy = 0;
   285→  for (let i = 0; i < n; i++) {
   286→    const a = xs[i] - mx, b = ys[i] - my;
   287→    num += a * b; dx += a * a; dy += b * b;
   288→  }
   289→  const denom = Math.sqrt(dx) * Math.sqrt(dy);
   290→  return denom === 0 ? 0 : num / denom;
   291→}

   310→  for (const o of orders) {

   312→    totalProfit += o.profit;
   313→    totalShipping += o.shippingCost;

   316→    customers.add(o.customerName);

   318→    if (!yearly[y]) yearly[y] = { revenue: 0, profit: 0, orders: 0 };

   445→  const catKeys = Array.from(catMap.keys());
   446→  const segCatMap = new Map<string, Record<string, number>>();
   447→  for (const o of orders) {
   448→    if (!segCatMap.has(o.customerSegment)) segCatMap.set(o.customerSegment, Object.fromEntries(catKeys.map((c) => [c, 0])));
   449→    segCatMap.get(o.customerSegment)![o.productCategory] += o.totalSales;
   450→  }
   451→  const segment_category = Array.from(segCatMap.entries()).map(([seg, vals]) => ({
   452→    segment: seg, values: Object.fromEntries(Object.entries(vals).map(([k, v]) => [k, fcur(v)])),
   453→  }));
   454→
   455→  // Discount scatter sample (sample 200 items)
   456→  const sample = [...orders].sort(() => Math.random() - 0.5).slice(0, 200);
   457→  const discount_scatter = sample.map((o) => ({
   458→    discount: o.discountPercent,
   459→    profit_margin: o.totalSales > 0 ? round((o.profit / o.totalSales) * 100, 2) : 0,
   460→    total_sales: fcur(o.totalSales), category: o.productCategory, product: o.productName,
   461→  }));
   462→
   463→  // Discount brackets
   464→  const drMap = new Map<string, { orders: number; revenue: number; profit: number; marginSum: number }>();
   465→  for (const o of orders) {
   466→    const r = discountRange(o.discountPercent);
   467→    if (!drMap.has(r)) drMap.set(r, { orders: 0, revenue: 0, profit: 0, marginSum: 0 });
   468→    const e = drMap.get(r)!;
   469→    e.orders += 1; e.revenue += o.totalSales; e.profit += o.profit;
   470→    if (o.totalSales > 0) e.marginSum += (o.profit / o.totalSales) * 100;
   471→  }
   472→  const drOrder = ["No Discount", "1-10%", "11-20%", "21-35%"];
   473→  const discount_ranges = drOrder.map((r) => {
   474→    const d = drMap.get(r)!;
   475→    return { range: r, orders: d.orders, avg_revenue: fcur(d.revenue / d.orders),
   476→      avg_profit: fcur(d.profit / d.orders), avg_margin: round(d.marginSum / d.orders, 2) };
   477→  }).filter((d) => d.orders > 0);
   478→
   479→  // Payment methods
   480→  const payMap = new Map<string, { orders: number; revenue: number }>();
   481→  for (const o of orders) {
   482→    if (!payMap.has(o.paymentMethod)) payMap.set(o.paymentMethod, { orders: 0, revenue: 0 });
   483→    const e = payMap.get(o.paymentMethod)!;
   484→    e.orders += 1; e.revenue += o.totalSales;
   485→  }
   486→  const totalPayOrders = Array.from(payMap.values()).reduce((s, v) => s + v.orders, 0);
   487→  const payment = Array.from(payMap.entries())
   488→    .map(([k, v]) => ({
   489→      method: k, orders: v.orders, revenue: fcur(v.revenue),
   490→      share: round((v.orders / totalPayOrders) * 100, 1), avg_order: fcur(v.revenue / v.orders),
   491→    })).sort((a, b) => b.orders - a.orders);
   492→
   493→  const shipping = region.map((r) => ({
   494→    region: r.region, avg_shipping: r.avg_shipping, profit_margin: r.profit_margin,
   495→  }));
   496→
   497→  // Pearson correlation matrix
   498→  const CORR_FEATURES = ["Quantity", "Unit_Price", "Discount_Percent", "Total_Sales", "Shipping_Cost", "Profit", "Profit_Margin"];
   499→  const corrCols: Record<string, number[]> = {};
   500→  for (const f of CORR_FEATURES) corrCols[f] = [];
   501→  for (const o of orders) {
   502→    corrCols.Quantity.push(o.quantity);
   503→    corrCols.Unit_Price.push(o.unitPrice);
   504→    corrCols.Discount_Percent.push(o.discountPercent);
   505→    corrCols.Total_Sales.push(o.totalSales);
   506→    corrCols.Shipping_Cost.push(o.shippingCost);
   507→    corrCols.Profit.push(o.profit);
   508→    corrCols.Profit_Margin.push(o.totalSales > 0 ? (o.profit / o.totalSales) * 100 : 0);
   509→  }
   510→  const correlation = CORR_FEATURES.map((a) => {
   511→    const row: Record<string, string | number> = { feature: a };
   512→    for (const b of CORR_FEATURES) row[b] = round(pearson(corrCols[a], corrCols[b]), 2);
   513→    return row;
   514→  });
   515→
   516→  // Day of week patterns
   517→  const dowMap = new Map<string, { orders: number; revenue: number }>();
   518→  for (const d of DOW_ORDER) dowMap.set(d, { orders: 0, revenue: 0 });
   519→  for (const o of orders) {
   520→    const d = DOW_ORDER[o.orderDate.getUTCDay()];
   521→    const e = dowMap.get(d)!;
   522→    e.orders += 1; e.revenue += o.totalSales;
   523→  }
   524→  const dow = DOW_ORDER.slice(1).concat([DOW_ORDER[0]]).map((d) => {
   525→    const e = dowMap.get(d)!;
   526→    return { day: d, orders: e.orders, avg_revenue: e.orders > 0 ? fcur(e.revenue / e.orders) : 0 };
   527→  });
   528→
   529→  return {
   530→    kpis, monthly, yoy, heatmap, region, country, category, top_products,
   531→    segment, segment_category, discount_scatter, discount_ranges, payment,
   532→    shipping, correlation, correlation_features: CORR_FEATURES, dow,
   533→  };
   534→}
   535→
   536→export type RecentOrdersParams = {
   537→  search?: string; category?: string; region?: string;
   538→  payment?: string; segment?: string; limit?: number; offset?: number;
   539→};
   540→
   541→export async function getRecentOrders(params: RecentOrdersParams) {
   542→  const { search, category, region, payment, segment, limit = 25, offset = 0 } = params;
   543→  const where: Record<string, unknown> = {};
   544→  if (category && category !== "all") where.productCategory = category;
   545→  if (region && region !== "all") where.region = region;
   546→  if (payment && payment !== "all") where.paymentMethod = payment;
   547→  if (segment && segment !== "all") where.customerSegment = segment;
   548→  if (search) {
   549→    where.OR = [
   550→      { id: { contains: search } },
   551→      { customerName: { contains: search } },
   552→      { productName: { contains: search } },
   553→      { country: { contains: search } },
   554→    ];
   555→  }
   556→  const [total, rows] = await Promise.all([
   557→    db.order.count({ where }),
   558→    db.order.findMany({ where, orderBy: { orderDate: "desc" }, take: limit, skip: offset }),
   559→  ]);
   560→  return {
   561→    total,
   562→    rows: rows.map((o) => ({
   563→      id: o.id, date: o.orderDate.toISOString().slice(0, 10),
   564→      customer: o.customerName, country: o.country, region: o.region,
   565→      product: o.productName, category: o.productCategory, quantity: o.quantity,
   566→      total_sales: round(o.totalSales, 2), profit: round(o.profit, 2),
   567→      discount_percent: o.discountPercent, payment_method: o.paymentMethod,
   568→      customer_segment: o.customerSegment,
   569→    })),
   570→  };
   571→}

   771→
   772→### 📄 [`scripts/seed-orders.ts`](file:///c:/Users/manga/Downloads/archive/scripts/seed-orders.ts)
   773→Generates 2,000 synthetic transactions spanning 2023–2025 in batch chunks of 250 using Prisma `createMany`:
   774→
   775→```typescript
   776→import { PrismaClient } from "@prisma/client";
   777→import {
   778→  COUNTRIES, REGIONS, CATEGORIES, PRODUCTS, CUSTOMER_SEGMENTS, PAYMENT_METHODS,
   779→  REGION_SHIP_BASE, CATEGORY_MARGIN, SEGMENT_DISCOUNT, MONTH_WEIGHTS,
   780→  FIRST_NAMES, LAST_NAMES, weightedPick, makeRng,
   781→  type Region, type Category, type CustomerSegment,
   782→} from "../src/lib/ecommerce-domain";
   783→
   784→const db = new PrismaClient();
   785→const N_TXNS = 2000;
   786→const YEARS = [2023, 2024, 2025];
   787→
   788→const rand = (rng: () => number, min: number, max: number) => min + rng() * (max - min);
   789→const randInt = (rng: () => number, lo: number, hi: number) => Math.floor(rand(rng, lo, hi + 1));
   790→const pick = <T,>(arr: readonly T[], rng: () => number): T => arr[Math.floor(rng() * arr.length)];
   791→
   792→function generateOrder(i: number, rng: () => number) {
   793→  const month = weightedPick(Array.from({ length: 12 }, (_, k) => k + 1), MONTH_WEIGHTS);
   794→  const year = pick(YEARS, rng);
   795→  const day = randInt(rng, 1, 28);
   796→  const orderDate = new Date(year, month - 1, day);
   797→
   798→  const region = weightedPick(REGIONS as unknown as Region[], [34, 28, 22, 9, 7]);
   799→  const country = pick(COUNTRIES[region], rng);
   800→  const category = weightedPick(CATEGORIES as unknown as Category[], [40, 28, 20, 12]);
   801→  const product = pick(PRODUCTS[category], rng);
   802→  const unitPrice = product.price;
   803→  const quantity = weightedPick([1, 2, 3, 4, 5], [45, 28, 14, 8, 5]);
   804→  const segment = weightedPick(CUSTOMER_SEGMENTS as unknown as CustomerSegment[], [55, 30, 15]);
   805→
   806→  let [dLo, dHi] = SEGMENT_DISCOUNT[segment];
   807→  if (unitPrice > 200) dLo += 2;
   808→  const discountPercent = rng() > 0.15 ? Math.round(rand(rng, dLo, dHi) * 10) / 10 : 0;
   809→
   810→  const gross = unitPrice * quantity;
   811→  const discountAmount = (gross * discountPercent) / 100;
   812→  const totalSales = Math.round((gross - discountAmount) * 100) / 100;
   813→
   814→  const baseMargin = CATEGORY_MARGIN[category];
   815→  const shippingCost = Math.round((REGION_SHIP_BASE[region] + rand(rng, -3, 5)) * 100) / 100;
   816→  const profit = Math.round((baseMargin * gross - discountAmount - shippingCost) * 100) / 100;
   817→
   818→  const payment = weightedPick(PAYMENT_METHODS as unknown as string[], [55, 25, 12, 8]);
   819→  const customer = `${pick(FIRST_NAMES, rng)} ${pick(LAST_NAMES, rng)}`;
   820→  const yearIdx = YEARS.indexOf(year);
   821→  const id = `ORD-${yearIdx * 10000 + i + 1}`;
   822→
   823→  return {
   824→    id, orderDate, customerName: customer, customerSegment: segment,
   825→    country, region, productCategory: category, productName: product.name,
   826→    quantity, unitPrice, discountPercent, shippingCost, totalSales,
   827→    profit, paymentMethod: payment,
   828→  };
   829→}
   830→
   831→async function main() {
   832→  const existing = await db.order.count();
   833→  if (existing > 0) {
   834→    console.log(`Database already has ${existing} orders. Skipping seed.`);
   835→    return;
   836→  }
   837→  console.log(`Seeding ${N_TXNS} synthetic orders...`);
   838→  const rng = makeRng(42);
   839→  const CHUNK = 250;
   840→  for (let chunkStart = 0; chunkStart < N_TXNS; chunkStart += CHUNK) {
   841→    const batch = [];
   842→    for (let i = chunkStart; i < Math.min(chunkStart + CHUNK, N_TXNS); i++) {
   843→      batch.push(generateOrder(i, rng));
   844→    }
   845→    await db.order.createMany({ data: batch });
   846→    console.log(`  Inserted orders ${chunkStart + 1}–${chunkStart + batch.length}`);
   847→  }
   848→  const final = await db.order.count();
   849→  console.log(`✓ Seed complete. Database now has ${final} orders.`);
   850→}
   851→
   852→main().catch((e) => { console.error("Seed failed:", e); process.exit(1); })
   853→  .finally(async () => { await db.$disconnect(); });
   854→```
